    <script>
        let carrito = [];

        function agregarAlCarro(nombre, precio) {
            let item = carrito.find(p => p.name === nombre);
            if (item) {
                item.qty++;
            } else {
                carrito.push({ name: nombre, price: precio, qty: 1 });
            }
            actualizarCarrito();
        }

        function actualizarCarrito() {
            let totalItems = 0;
            let totalPrice = 0;

            carrito.forEach(item => {
                totalItems += item.qty;
                totalPrice += (item.price * item.qty);
            });

            document.getElementById('cart-count').innerText = totalItems;
            document.getElementById('cart-total').innerText = '$' + totalPrice.toFixed(2);
        }

        function enviarPedidoWhatsApp(e) {
            if (carrito.length === 0) {
                e.preventDefault();
                alert('Por favor selecciona al menos un plato de nuestro menú antes de ordenar.');
                return;
            }

            let mensaje = "Hola, *Le Maison*! 🍷 Deseo realizar el siguiente pedido:%0A%0A";
            let total = 0;

            carrito.forEach(item => {
                let subtotal = item.price * item.qty;
                mensaje += `• ${item.qty}x ${item.name} ($${subtotal.toFixed(2)})%0A`;
                total += subtotal;
            });

            mensaje += `%0A*Total a Pagar:* $${total.toFixed(2)}%0A%0AQuedo atento a la confirmación de mi reserva/pedido.`;

            let telefono = "584125952360";
            let url = `https://wa.me/${telefono}?text=${mensaje}`;

            document.getElementById('whatsapp-btn').href = url;
        }
    </script>
